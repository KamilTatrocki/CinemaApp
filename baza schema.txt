STRUKTURA BAZY DANYCH - APKA KINOWA
-----------------------------------

1. Tabela: Movies (Filmy)
   - id: INT (PK, Auto Increment)
   - title: VARCHAR(255)
   - description: TEXT
   - duration_minutes: INT
   - release_year: INT
   - rating: DECIMAL(3,1)

2. Tabela: Cinemas (Kina)
   - id: INT (PK, Auto Increment)
   - name: VARCHAR(100)
   - city: VARCHAR(100)
   - address: VARCHAR(255)

3. Tabela: Halls (Sale Kinowe)
   - id: INT (PK, Auto Increment)
   - cinema_id: INT (FK -> Cinemas.id)
   - name: VARCHAR(50) (np. "Sala 1", "Sala IMAX")

4. Tabela: Seats (Miejsca w salach)
   - id: INT (PK, Auto Increment)
   - hall_id: INT (FK -> Halls.id)
   - row_label: VARCHAR(5) (np. "A", "B")
   - seat_number: INT
   - type: VARCHAR(20) (np. "Normal", "VIP")

5. Tabela: Screenings (Seanse)
   - id: INT (PK, Auto Increment)
   - movie_id: INT (FK -> Movies.id)
   - hall_id: INT (FK -> Halls.id)
   - start_time: DATETIME
   - base_price: DECIMAL(10,2)

6. Tabela: Users (Użytkownicy)
   - id: INT (PK, Auto Increment)
   - email: VARCHAR(255) (Unique)
   - password_hash: VARCHAR(255)
   - full_name: VARCHAR(100)

7. Tabela: Ticket_Types (Typy biletów)
   - id: INT (PK, Auto Increment)
   - name: VARCHAR(50) (np. "Adult", "Student", "Senior")
   - discount_percentage: INT (np. 0, 20, 50)

8. Tabela: Reservations (Rezerwacje/Zamówienia)
   - id: INT (PK, Auto Increment)
   - user_id: INT (FK -> Users.id)
   - screening_id: INT (FK -> Screenings.id)
   - total_price: DECIMAL(10,2)
   - created_at: TIMESTAMP

9. Tabela: Tickets (Bilety - konkretne miejsca)
   - id: INT (PK, Auto Increment)
   - reservation_id: INT (FK -> Reservations.id)
   - seat_id: INT (FK -> Seats.id)
   - ticket_type_id: INT (FK -> Ticket_Types.id)
   - qr_code_token: VARCHAR(255)

10. Tabela: Promotions (Promocje barowe)
    - id: INT (PK, Auto Increment)
    - title: VARCHAR(100)
    - description: TEXT
